# 4. РАБОТА С KSQLDB (STREAMS и TABLES)
# ============================================

cat > /tmp/ksql_processing.sql << 'EOF'
-- ============================================
-- БЛОК 1: НАСТРОЙКА СЕССИИ
-- SET 'auto.offset.reset' действует ТОЛЬКО в текущей сессии ksqlDB CLI,
-- поэтому повторяем его в начале КАЖДОГО файла/вызова (на сервере уже задан
-- ksql.streams.auto.offset.reset=earliest, но в новой сессии без SET
-- персистентный запрос мог бы стартовать с конца топика и пропустить
-- снапшот Debezium из лабы 3.2).
-- ============================================
SET 'auto.offset.reset' = 'earliest';

-- ============================================
-- БЛОК 2: ЧТЕНИЕ CDC-ТОПИКА ИЗ ЛАБЫ 3.2 (STREAM raw_orders)
-- Топик postgres-server.public.orders пишет Debezium из лабы 3.2.
-- Значение — JSON без схемы (JsonConverter, schemas.enable=false), обёртка
-- события: {"before":..., "after":{...}, "source":{...}, "op":"c","ts_ms":...}
-- Ключи JSON — в нижнем регистре, поэтому колонки/STRUCT объявляем так же.
-- price объявлен DOUBLE: у Debezium-коннектора в 3.2 должен быть задан
-- decimal.handling.mode=double (правка конфигурации коннектора в лабе 3.2),
-- иначе NUMERIC придёт строкой base64 и разбор не сработает. В схеме
-- используем DOUBLE, а не голый DECIMAL (такого типа в ksqlDB нет).
-- ============================================

-- Сначала посмотрим сообщение (опционально, для понимания структуры):
-- PRINT 'postgres-server.public.orders' FROM BEGINNING LIMIT 1;

CREATE STREAM raw_orders (
  op VARCHAR,
  before STRUCT<id INT, order_number VARCHAR, customer_name VARCHAR, product VARCHAR, quantity INT, price DOUBLE>,
  after  STRUCT<id INT, order_number VARCHAR, customer_name VARCHAR, product VARCHAR, quantity INT, price DOUBLE>,
  ts_ms BIGINT
) WITH (
  KAFKA_TOPIC='postgres-server.public.orders',
  VALUE_FORMAT='JSON'
);

-- ============================================
-- БЛОК 3: ЧИСТЫЙ ПОТОК ЗАКАЗОВ (STREAM orders_stream)
-- Извлекаем полезную нагрузку AFTER; события удаления (after = NULL) отбрасываем.
-- ============================================
CREATE STREAM orders_stream
WITH (KAFKA_TOPIC='orders_clean', VALUE_FORMAT='JSON') AS
SELECT
  after->id AS id,
  after->order_number AS order_number,
  after->customer_name AS customer_name,
  after->product AS product,
  after->quantity AS quantity,
  after->price AS price
FROM raw_orders
WHERE after IS NOT NULL
EMIT CHANGES;

-- ============================================
-- БЛОК 4: ОБОГАЩЕНИЕ (STREAM enriched_orders)
-- Итоговая сумма заказа и категория товара по цене
-- ============================================
CREATE STREAM enriched_orders AS
SELECT
  id,
  order_number,
  customer_name,
  product,
  quantity,
  price,
  (quantity * price) AS total_price,
  CASE
    WHEN price > 100000 THEN 'Premium'
    WHEN price > 50000 THEN 'Standard'
    ELSE 'Budget'
  END AS category
FROM orders_stream
EMIT CHANGES;

-- ============================================
-- БЛОК 5: ОКОННАЯ АГРЕГАЦИЯ (TABLE revenue_30s, TUMBLING 30 СЕКУНД)
-- Канонический вид: WINDOWSTART/WINDOWEND движок добавляет в ключ окна
-- автоматически — их НЕЛЬЗЯ указывать в GROUP BY, только в SELECT.
-- Топик результата задаём явно (нижний регистр), в него будет писать sink.
-- ============================================
CREATE TABLE revenue_30s
WITH (KAFKA_TOPIC='revenue_30s') AS
SELECT
  category,
  SUM(total_price) AS total_revenue,
  COUNT(*) AS order_count,
  WINDOWSTART AS window_start,
  WINDOWEND AS window_end
FROM enriched_orders
WINDOW TUMBLING (SIZE 30 SECONDS)
GROUP BY category
EMIT CHANGES;

-- ============================================
-- БЛОК 6: СТАТИСТИКА ПО КЛИЕНТАМ (TABLE client_stats)
-- TABLE хранит ТЕКУЩЕЕ (последнее по ключу customer_name) состояние
-- ============================================
CREATE TABLE client_stats AS
SELECT
  customer_name,
  COUNT(*) AS orders_count,
  SUM(total_price) AS spent_total
FROM enriched_orders
GROUP BY customer_name
EMIT CHANGES;

EOF

echo "🔨 Создаем Streams, Tables и агрегации..."
/opt/ksqldb/bin/ksql http://localhost:8088 --file /tmp/ksql_processing.sql

# Проверка количества обработанных заказов в ksqlDB (SELECT COUNT(*) с

# ============================================
# 5. SINK РЕЗУЛЬТАТОВ В OPENSEARCH (КОНЕЧНАЯ ТОЧКА)
# ============================================

# Используем ТОТ ЖЕ Aiven OpenSearch-коннектор, что и в лабе 3.2
# (io.aiven.kafka.connect.opensearch.OpensearchSinkConnector) — он уже лежит
# в /opt/kafka/connect-plugins. Создаём коннектор через REST ВНЕШНЕГО Kafka
# Connect из лабы 3.2 (REST http://localhost:8083) — так же, как в 3.2;
# отдельный (Confluent/Elasticsearch) sink не нужен и не совместим с курсом.
# Параметр называется "type" (не "type.name" эпохи ES 6.x), значение "_doc".
# Топик результата: revenue_30s — ksqlDB записал туда агрегаты из таблицы
# revenue_30s (имя топика мы задали явно в CTAS, см. блок 5 файла processing).
# Ключи оконных таблиц ksqlDB — НЕ JSON, поэтому для ключа берём
# StringConverter (иначе JsonConverter не разберёт ключ окна).

cat > /tmp/opensearch-revenue-sink.json << 'EOF'
{
  "name": "opensearch-revenue-sink",
  "config": {
    "connector.class": "io.aiven.kafka.connect.opensearch.OpensearchSinkConnector",
    "connection.url": "http://10.0.2.30:9200",
    "connection.username": "",
    "connection.password": "",
    "topics": "revenue_30s",
    "type": "_doc",
    "key.ignore": "true",
    "schema.ignore": "true",
    "behavior.on.null.values": "ignore",
    "behavior.on.malformed.documents": "warn",
    "drop.invalid.message": "true",
    "tasks.max": "1",
    "key.converter": "org.apache.kafka.connect.storage.StringConverter",
    "value.converter": "org.apache.kafka.connect.json.JsonConverter",
    "value.converter.schemas.enable": "false"
  }
}
EOF

echo "📤 Регистрируем Sink-коннектор OpenSearch во внешнем Connect (лаба 3.2)..."
curl -X POST -H "Content-Type: application/json" \
  --data @/tmp/opensearch-revenue-sink.json \
  http://localhost:8083/connectors

# Проверяем статус созданного коннектора
curl -s http://localhost:8083/connectors/opensearch-revenue-sink/status | jq .state
# Ожидаем "RUNNING"

curl -s http://localhost:8083/connectors | jq .
# Ожидаем: коннекторы из лабы 3.2 (postgres-source-connector,

echo "=========================================="

# Вставляем заказ в PostgreSQL (10.0.2.31, БД labdb).
# ВНИМАНИЕ: пароль labpassword используется только для учебного контура.
# При ПОВТОРНОМ прохождении лабы заказ ORD-KAFKA-LAB может уже существовать
# (у order_number UNIQUE-ограничение) — тогда возьмите другой номер, например
# ORD-KAFKA-LAB-2, и в шаге 6.3 используйте его же.
echo "Добавляем новый заказ в PostgreSQL..."

PGPASSWORD=labpassword psql -h 10.0.2.31 -U labuser -d labdb -c \
  "INSERT INTO orders (order_number, customer_name, product, quantity, price) VALUES \
   ('ORD-KAFKA-LAB', 'Студент курса', 'Курс по ksqlDB', 1, 99999.00);"
echo "✅ Заказ добавлен. Debezium (лаба 3.2) пишет событие в Kafka."

# Детерминированная проверка CDC (вместо sleep+kill): опрашиваем топик с
# начала, пока не найдём наше событие (order_number уникален — коллизий нет).
# Каждая попытка сама завершается по таймауту или после прочтения сообщений.
# Максимум ~15 попыток; между ними пауза 2 секунды.
FOUND=0
for attempt in $(seq 1 15); do
  FOUND=$(/opt/kafka/bin/kafka-console-consumer.sh \
    --bootstrap-server localhost:9092 \
    --topic postgres-server.public.orders \
    --from-beginning \
    --max-messages 1000 \
    --timeout-ms 3000 2>/dev/null | grep -c ORD-KAFKA-LAB)
  [ "$FOUND" -gt 0 ] && break
  echo "Попытка $attempt: события с нашим заказом ещё нет, ждём 2 c..."
  sleep 2
done

if [ "$FOUND" -gt 0 ]; then
  echo "✅ Событие CDC с ORD-KAFKA-LAB найдено в топике."
else
  echo "❌ Событие не найдено: проверьте, что коннектор из 3.2 жив (см. раздел 3),"
  echo "   а заказ действительно вставлен (UNIQUE по order_number не нарушен)."
fi

echo ""
echo "=========================================="
echo "6.3. ЧТЕНИЕ ИЗ KSQLDB: PUSH- И PULL-ЗАПРОСЫ"
echo "=========================================="

# --- Push-запрос (EMIT CHANGES) к STREAM ---
# Подписывается на события стрима и печатает новые строки по мере их
# появления; в конце запроса указывается EMIT CHANGES, при необходимости —
# LIMIT N для остановки. SET 'auto.offset.reset' нужен в начале сессии,
# чтобы запрос увидел уже накопленные события, а не только новые.
cat > /tmp/ksql_push.sql << 'EOF'
SET 'auto.offset.reset' = 'earliest';
SELECT order_number, category, total_price
FROM enriched_orders
EMIT CHANGES
LIMIT 3;
EOF

echo "📊 Push-запрос к STREAM enriched_orders (выведет 3 строки и завершится):"
/opt/ksqldb/bin/ksql http://localhost:8088 --file /tmp/ksql_push.sql

echo ""
echo "📊 Pull-запрос к TABLE client_stats по ключу:"
# --- Pull-запрос к TABLE по ключу ---
# Табличный pull-запрос возвращает СНИМОК состояния и завершается сам, но
# обязан содержать WHERE с равенством по ВСЕМ ключевым колонкам таблицы.
