Лабораторная работа 3.3: Настройка мониторинга кластера Kafka с Prometheus и алертами на лаг
Цель работы
Настроить комплексную систему мониторинга для кластера Apache Kafka с использованием Prometheus, Alertmanager и Grafana. Научиться собирать метрики с брокеров Kafka через kafka-exporter, настраивать алерты на критически важные показатели (особенно на лаг консумеров) и визуализировать метрики в Grafana.

Часть 2: Настройка сбора метрик с Kafka кластера
Шаг 2.1: Установка Node Exporter на нодах Kafka
# На каждой ноде Kafka (kafka1, kafka2, kafka3) выполняем:

# 1. Скачиваем Node Exporter
wget https://github.com/prometheus/node_exporter/releases/download/v1.6.1/node_exporter-1.6.1.linux-amd64.tar.gz

# 2. Распаковываем
tar -xzf node_exporter-1.6.1.linux-amd64.tar.gz
sudo mv node_exporter-1.6.1.linux-amd64/node_exporter /usr/local/bin/
sudo chmod +x /usr/local/bin/node_exporter

# 3. Создаем systemd сервис
sudo tee /etc/systemd/system/node-exporter.service << 'EOF'
[Unit]
Description=Node Exporter
After=network.target

[Service]
Type=simple
User=prometheus
Group=prometheus
ExecStart=/usr/local/bin/node_exporter \
  --collector.systemd \
  --collector.processes \
  --collector.filesystem \
  --collector.netstat \
  --collector.meminfo \
  --collector.cpu

[Install]
WantedBy=multi-user.target
EOF

# 4. Создаем пользователя для Node Exporter
sudo useradd -rs /bin/false prometheus

# 5. Запускаем Node Exporter
sudo systemctl daemon-reload
sudo systemctl enable node-exporter
sudo systemctl start node-exporter

# 6. Проверяем статус
sudo systemctl status node-exporter

# 7. Открываем порт 9100 в ufw для Prometheus (скрейп с 10.0.2.30)
# В лабораторной 1 на брокерах в ufw открыты только порты 9092:9095.
# Без этого правила Prometheus с 10.0.2.30 не достучится до node_exporter,
# и алерты на CPU/память/диск останутся без данных («мёртвые»).
# Порт 8083 (Kafka Connect) здесь не открываем: его джоб в prometheus.yml
# закомментирован — у Kafka Connect нет HTTP /metrics (см. Шаг 3.2).
sudo ufw allow from 10.0.2.30 to any port 9100 proto tcp
sudo ufw status | grep 9100

Часть 3: Развертывание стека мониторинга через Docker Compose

cd ~/kafka-monitoring

cat > docker-compose.yml << 'EOF'

  prometheus_data: {}
  grafana_data: {}
  alertmanager_data: {}

      - '--config.file=/etc/prometheus/prometheus.yml'
      - '--storage.tsdb.path=/prometheus'
      - '--storage.tsdb.retention.time=30d'
      - '--web.enable-lifecycle'
    ports:
      - "9090:9090"
    networks:
      - monitoring

  alertmanager:
    image: prom/alertmanager:v0.28.0  # проверьте актуальные теги при подготовке курса
    container_name: alertmanager
    restart: unless-stopped
    volumes:
      - ./alertmanager/alertmanager.yml:/etc/alertmanager/alertmanager.yml
      - alertmanager_data:/alertmanager
    ports:
      - "9093:9093"
    networks:
      - monitoring

  grafana:
    image: grafana/grafana:11.3.0  # проверьте актуальные теги при подготовке курса
    container_name: grafana
    restart: unless-stopped
    environment:
      - GF_SECURITY_ADMIN_PASSWORD=admin123
      # GF_INSTALL_PLUGINS=grafana-piechart-panel удалён: панель pie chart
      # встроена в Grafana начиная с версии 8.3 (core-плагин), отдельная

    volumes:
      - grafana_data:/var/lib/grafana
      - ./grafana/provisioning:/etc/grafana/provisioning
      - ./grafana/dashboards:/var/lib/grafana/dashboards
    ports:
      - "3000:3000"
    networks:
      - monitoring

  kafka-exporter:
    image: danielqsj/kafka-exporter:v1.6.0  # проверьте актуальные теги при подготовке курса; проверить совместимость v1.6.0 с Kafka 4.1.1 (KRaft)
    container_name: kafka-exporter
    restart: unless-stopped
    command:
      - '--kafka.server=10.0.2.20:9092'
      - '--kafka.server=10.0.2.21:9092'
      - '--kafka.server=10.0.2.22:9092'
      - '--web.listen-address=:9308'
      - '--log.level=info'
    ports:
      - "9308:9308"
    networks:

Шаг 3.2: Настройка конфигурации Prometheus
# Создаем конфигурационный файл Prometheus
mkdir -p prometheus
cat > prometheus/prometheus.yml << 'EOF'
global:
  scrape_interval: 15s
  evaluation_interval: 15s
  external_labels:
    monitor: 'kafka-cluster'

rule_files:
  - "alerts.yml"

alerting:
  alertmanagers:
    - static_configs:
        - targets:
            - alertmanager:9093

scrape_configs:
  # Prometheus сам по себе
  - job_name: 'prometheus'
    static_configs:
      - targets: ['localhost:9090']

  # Node Exporters на нодах Kafka
  - job_name: 'node-exporter-kafka'
    static_configs:
      - targets:
          - '10.0.2.20:9100'
          - '10.0.2.21:9100'
          - '10.0.2.22:9100'
    relabel_configs:
      - source_labels: [__address__]
        target_label: instance
        regex: '([^:]+):\d+'
        replacement: '${1}'

  # Kafka Exporter для метрик Kafka
  - job_name: 'kafka-exporter'
    static_configs:
      - targets: ['kafka-exporter:9308']
    metrics_path: '/metrics'

  # Kafka Connect: джоб закомментирован. У Kafka Connect НЕТ HTTP-эндпоинта
  # /metrics на REST-порту 8083 (там admin API; метрики отдаются только через
  # JMX), поэтому скрейп сюда бессмыслен — таргет был бы вечно DOWN.
  # Для этой лабораторной работы достаточно kafka-exporter и node_exporter;
  # если мониторинг Connect всё же нужен — настройте отдельно jmx_exporter
  # (и откройте его порт в ufw на ноде Kafka).
  # - job_name: 'kafka-connect'
  #   static_configs:
  #     - targets: ['10.0.2.20:8083']
  #   metrics_path: '/metrics'
EOF

Шаг 3.3: Настройка алертов для Kafka
# Создаем файл с правилами алертов
cat > prometheus/alerts.yml << 'EOF'
groups:
  - name: kafka_alerts
    rules:
      # Алерт на высокий consumer lag
      # Метрика danielqsj/kafka-exporter называется kafka_consumergroup_lag
      # (лейблы consumergroup/topic/partition). Лаг считается по каждой
      # партиции, поэтому агрегируем по группе и топику — иначе алерт
      # срабатывал бы на лаг отдельной партиции.
      - alert: HighConsumerLag
        expr: |
          sum by (consumergroup, topic) (kafka_consumergroup_lag) > 1000
        for: 1m  # для лабораторной работы не ждём 5 минут (иначе проверка алерта в Части 6 затянется)
        labels:
          severity: critical
          service: kafka
        annotations:
          summary: "Высокий лаг консумера"
          description: "Группа {{ $labels.consumergroup }} имеет суммарный лаг {{ $value }} сообщений (топик {{ $labels.topic }})"

      # Алерт на недоступность брокера Kafka
      # Внимание: правило kafka_brokers < 3 НЕ надёжно — остановленный брокер
      # часто остаётся в metadata кластера, и kafka-exporter продолжает
      # сообщать 3. Падение ЦЕЛОЙ ноды надёжнее ловить через node_exporter:
      # см. правило KafkaNodeDown ниже.
      - alert: KafkaBrokerDown
        expr: |
          kafka_brokers < 3
        for: 1m
        labels:
          severity: critical
          service: kafka
        annotations:
          summary: "Брокер Kafka недоступен"
          description: "Только {{ $value }} брокеров доступно из 3"

      # Алерт на недоступность нод кластера Kafka (node_exporter не отвечает).
      # Имя job взято из prometheus.yml: node-exporter-kafka. Если вы
      # переименовали job в prometheus.yml — поправьте имя и здесь.
      - alert: KafkaNodeDown
        expr: |
          up{job="node-exporter-kafka"} == 0
        for: 1m
        labels:
          severity: warning
          service: node
        annotations:
          summary: "Нода Kafka недоступна: {{ $labels.instance }}"
          description: "Prometheus не получает метрики node_exporter с {{ $labels.instance }}"

      # Алерт на высокую загрузку CPU
      - alert: HighCPUUsage
        expr: |
          100 - (avg(rate(node_cpu_seconds_total{mode="idle"}[5m])) by (instance) * 100) > 80
        for: 5m
        labels:
          severity: warning
          service: node
        annotations:
          summary: "Высокая загрузка CPU на {{ $labels.instance }}"
          description: "Загрузка CPU составляет {{ $value }}%"

      # Алерт на нехватку памяти
      - alert: HighMemoryUsage
        expr: |
          (node_memory_MemTotal_bytes - node_memory_MemAvailable_bytes) / node_memory_MemTotal_bytes * 100 > 85
        for: 5m
        labels:
          severity: warning
          service: node
        annotations:
          summary: "Высокое использование памяти на {{ $labels.instance }}"
          description: "Используется {{ $value | humanizePercentage }} памяти"

      # Алерт на нехватку дискового пространства
      - alert: HighDiskUsage
        expr: |
          (node_filesystem_size_bytes{mountpoint="/"} - node_filesystem_free_bytes{mountpoint="/"}) / node_filesystem_size_bytes{mountpoint="/"} * 100 > 85
        for: 5m
        labels:
          severity: warning
          service: node
        annotations:
          summary: "Высокое использование диска на {{ $labels.instance }}"
          description: "Используется {{ $value | humanizePercentage }} дискового пространства"

      # Алерт на сам мониторинг: если цель скрейпа (экспортёр или сам
      # Prometheus) недоступна — сообщаем об этом явно. Без такого правила
      # «молчание» метрик можно принять за отсутствие проблем.
      - alert: MonitoringTargetsDown
        expr: |
          up{job=~"prometheus|kafka-exporter|node-exporter-kafka"} == 0
        for: 2m
        labels:
          severity: warning
          service: monitoring
        annotations:
          summary: "Цель мониторинга недоступна: {{ $labels.job }} ({{ $labels.instance }})"
          description: "Prometheus не получает метрики с джоба {{ $labels.job }} ({{ $labels.instance }})"
EOF

Шаг 3.4: Настройка Alertmanager
# Создаем конфигурацию Alertmanager
mkdir -p alertmanager
cat > alertmanager/alertmanager.yml << 'EOF'
# Нейтральная конфигурация для лабораторной работы: реальная доставка
# уведомлений (почта/вебхуки) здесь НЕ настраивается и НЕ проверяется.
#
# Чтобы получать реальные письма, добавьте в этот файл свой SMTP, например:
# global:
#   smtp_smarthost: '<ваш_smtp_сервер>:587'
#   smtp_from: '<ваш_email>'
#   smtp_auth_username: '<ваш_email>'
#   smtp_auth_password: '<ваш_пароль>'
#   smtp_require_tls: true
# ...
# receivers:
#   - name: 'webhook-null'
#     email_configs:
#       - to: '<ваш_email>'
#         send_resolved: true

route:
  group_by: ['alertname', 'severity']
  group_wait: 10s
  group_interval: 10s
  repeat_interval: 1h
  receiver: 'webhook-null'

receivers:
  # Нейтральный receiver (без notifier'ов): алерты принимаются и «гаснут»
  # никуда не отправляясь. Приём проверяется через API Alertmanager (Часть 6).
  # При необходимости замените на свой вебхук, например:
  # - name: 'webhook-null'
  #   webhook_configs:
  #     - url: 'http://localhost:5000/alert'
  - name: 'webhook-null'
EOF

# Альтернативная конфигурация для тестирования
cat > alertmanager/alertmanager-test.yml << 'EOF'
route:
  receiver: 'null'

receivers:
  - name: 'null'
EOF

Шаг 4.3: Загрузка готовых дашбордов
# Создаем скрипт для загрузки дашбордов
cat > download-dashboards.sh << 'EOF'
#!/bin/bash

# Ревизии дашбордов зафиксированы (…/revisions/N): если скачанный дашборд
# несовместим с вашей версией Grafana — проверьте актуальную ревизию
# на grafana.com (ID 7589 / 1860 / 3662).

# Дашборд для Kafka Exporter (ID: 7589)
wget -O grafana/dashboards/kafka-exporter.json https://grafana.com/api/dashboards/7589/revisions/1/download

# Дашборд для Node Exporter (ID: 1860)
wget -O grafana/dashboards/node-exporter-full.json https://grafana.com/api/dashboards/1860/revisions/22/download

# Дашборд для Prometheus (ID: 3662)
wget -O grafana/dashboards/prometheus-overview.json https://grafana.com/api/dashboards/3662/revisions/1/download
EOF

chmod +x download-dashboards.sh
./download-dashboards.sh

Важно: дашборд «Kafka Exporter (ID: 7589)» использует переменные job и instance,
которые по умолчанию определяются через метрику kafka_consumergroup_current_offset.
Эта метрика у danielqsj/kafka-exporter СУЩЕСТВУЕТ и появляется, когда хотя бы одна
consumer group закоммитила свои offset'ы. Если на момент импорта таких групп нет,
переменные job и instance в верхней части дашборда остаются пустыми, а часть панелей —
без данных. Это НЕ ошибка конфигурации Prometheus или kafka-exporter: запустите
консьюмера с группой и закоммитьте offset'ы (Часть 6) — дашборд наполнится данными.

Основная проверка работы kafka-exporter (выполняется на 10.0.2.30 ПОСЛЕ запуска
стека в Части 5; ниже продублирована в шагах 5.2 и 6.1):
- команда
  curl -s http://localhost:9308/metrics | grep -E "kafka_consumergroup_lag|kafka_brokers" | head -5
  должна показать метрики kafka_brokers и kafka_consumergroup_lag;
- в Prometheus по запросу kafka_brokers должно отображаться значение 3;
- дашборд «Kafka Cluster Overview» (kafka-overview.json) должен показывать
  количество брокеров, сообщения в секунду и лаг консумеров.

# Исправляем источники данных в скачанных дашбордах
for file in grafana/dashboards/*.json; do
    if [ -f "$file" ]; then
        echo "Исправление источника данных в $file"
        sed -i 's/"datasource": "[^"]*"/"datasource": "Prometheus"/g' "$file"

Шаг 6.2: Детерминированный тест алерта HighConsumerLag
# Сценарий: consumer группы читает ТОЛЬКО новые сообщения (auto.offset.reset=
# latest, БЕЗ --from-beginning) и по ограничению --max-messages 100 завершается,
# успев прочитать лишь малую часть потока. Продюсер тем временем пишет 10000
# сообщений с ограниченным throughput → суммарный лаг группы гарантированно
# вырастает (end-offset ~10000 против закоммиченных ~100).

# --- Выполняется на ноде kafka1 ---
ssh kafka1

# Создаем тестовый топик (RF=3 — единообразие курса)
/opt/kafka/bin/kafka-topics.sh --create \
  --topic test-monitoring \
  --partitions 3 \
  --replication-factor 3 \
  --bootstrap-server localhost:9092

# 1) Запускаем consumer группы в фоне. Роль «медленного» consumer'а играет
#    ограничение --max-messages 100: он прочитает 100 сообщений и завершится,
#    а группа останется с закоммиченными offset'ами. Автокоммит включаем явно
#    (интервал 1 с), чтобы offset'ы гарантированно закоммитились, пока consumer жив.
/opt/kafka/bin/kafka-console-consumer.sh \
  --topic test-monitoring \
  --bootstrap-server localhost:9092 \
  --group test-monitoring-group \
  --max-messages 100 \
  --timeout-ms 60000 \
  --consumer-property enable.auto.commit=true \
  --consumer-property auto.commit.interval.ms=1000 \
  > /tmp/consumer-test-monitoring.log 2>&1 &

# 2) Даем consumer'у 5 секунд, чтобы он присоединился к группе
sleep 5

# 3) Запускаем producer: 10000 сообщений с throughput ~500/с (окно ~20 секунд).
#    Consumer уже «в группе», но успеет прочитать только свои 100 сообщений —
#    лаг группы будет расти всё время, пока пишет продюсер.
/opt/kafka/bin/kafka-producer-perf-test.sh \
  --topic test-monitoring \
  --num-records 10000 \
  --record-size 1000 \
  --throughput 500 \
  --producer-props bootstrap.servers=localhost:9092

# 4) Ждем завершения фонового consumer'а и смотрим, что он прочитал
wait
tail -3 /tmp/consumer-test-monitoring.log
# Ожидание: "Processed a total of 100 messages"

# Выходим с kafka1 (проверки ниже выполняются на 10.0.2.30)
exit

# --- Выполняется на 10.0.2.30 ---

# 5) Проверяем, что kafka-exporter видит лаг группы. Метрика называется
#    kafka_consumergroup_lag (лейбл consumergroup="test-monitoring-group"),
#    kafka-exporter слушает только на 10.0.2.30 — поэтому и curl здесь.
curl -s http://localhost:9308/metrics | grep "test-monitoring-group"
# Ожидание: строки вида
#   kafka_consumergroup_lag{consumergroup="test-monitoring-group",topic="test-monitoring",partition="0"} 3.3e3
# Сумма по трём партициям заметно больше порога 1000.

# 6) Ждем ~70 секунд и проверяем, что алерт HighConsumerLag перешёл в firing.
#    Правило имеет for: 1m — алерт переходит в firing только после минуты
#    непрерывного превышения порога (сначала он будет в состоянии pending).
sleep 70

# Проверяем в Prometheus
curl -s "http://localhost:9090/api/v1/alerts" | jq '.data.alerts[] | select(.labels.alertname == "HighConsumerLag") | {state, labels, annotations}'
# Ожидание: "state": "firing"

# Проверяем в Alertmanager (HTTP API v2)
curl -s "http://localhost:9093/api/v2/alerts" | jq '.[] | select(.labels.alertname == "HighConsumerLag") | {status: .status.state, labels}'
# Ожидание: "status": "active" — алерт принят Alertmanager'ом

# 7) Проверяем resolved: даем группе «догнать». Consumer той же группы
#    продолжит с закоммиченного offset'а и дочитает оставшиеся ~9900 сообщений —
#    лаг упадёт до нуля, алерт уйдёт из активных.
# --- Выполняется на ноде kafka1 ---
ssh kafka1

/opt/kafka/bin/kafka-console-consumer.sh \
  --topic test-monitoring \
  --bootstrap-server localhost:9092 \
  --group test-monitoring-group \
  --max-messages 100000 \
  --timeout-ms 10000 \
  --consumer-property enable.auto.commit=true \
  --consumer-property auto.commit.interval.ms=1000
# Consumer дочитает все накопленные сообщения и завершится сам (10 секунд
# без новых данных). Если алерт ещё не resolved — повторите запуск: consumer
# продолжит с последнего закоммиченного offset'а и дочитает остаток.

# Выходим с kafka1
exit

# --- Выполняется на 10.0.2.30 ---
# 8) Через ~30 секунд проверяем, что активных алертов HighConsumerLag нет:
sleep 30
curl -s "http://localhost:9090/api/v1/alerts" | jq '.data.alerts[] | select(.labels.alertname == "HighConsumerLag")'
# Ожидание: пустой результат (алерт resolved, убран из активных)
